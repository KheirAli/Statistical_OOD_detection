Assume
\begin{equation}
    p(x)\approx\mathcal{N}(\mu,\Sigma)
\end{equation} near $x^*$. Let the precision matrix be
\begin{equation}
    \Lambda=\Sigma^{-1}=Udiag(\lambda_k)U^T
\end{equation}
Let the noise distribution be
\begin{equation}
    \epsilon\sim\mathcal{N}(0,\sigma^2I)
\end{equation}
Denote for convenience
\begin{equation}
    \alpha=\sigma^{-2}
\end{equation}

\hfill \break
The conditional prior given all other coordinates are
\begin{equation}
    p_i:=p(x_i|x_{\sim i}^*)=\mathcal{N}(\mu_i^{cond},\frac{1}{\Lambda_{ii}})
\end{equation}
The metric that we want to use for out-of-distribution detection is defined as
\begin{equation}
    m_i^{(p)}=\Big\vert\frac{1}{2}\Big(\big[\Lambda(x^*-\mu)\big]^2-1\Big)\Big\vert
\end{equation}
This follows from
\begin{equation}
    m_i^{(p)}=\Big\vert\frac{1}{2}\Big(\Lambda_{ii}(x_i^*-\mu^{cond}_i)^2-1\Big)\Big\vert
\end{equation}
and
\begin{equation}
    \mu^{cond}_i=\mu_i-\frac{1}{\Lambda_{ii}}\Lambda_{i,\sim i}(x_{\sim i}^*-\mu_{\sim i})\Rightarrow\Lambda_{ii}(x_i-\mu_i^{cond})=\big[\Lambda(x^*-\mu)\big]_i
\end{equation}

\hfill \break
The distribution from which we sample from is given as
\begin{equation}
    q_\sigma:=\mathcal{N}(\mu_{denoised},\Sigma_{denoised})
\end{equation}
The mean is
\begin{equation}
    \mu_{denoised}=(I-B_{\alpha})\mu+B_{\alpha}x^*
\end{equation}
for
\begin{equation}
    B_\alpha=Udiag(\frac{\alpha}{\lambda_k+\alpha})U^T
\end{equation}
so
\begin{equation}
    \mu=\Big(I-B_\alpha\Big)^{-1}\Big(\mu_{denoised}-B_\alpha x^*\Big)
\end{equation}
The covariance is
\begin{equation}
    \Sigma_{denoised}=\Big(\Lambda+\alpha I\Big)^{-1}+\alpha\Big(\Lambda+\alpha I\Big)^{-2}
\end{equation}
Under this setting where the error is homoscedastic with variance $\sigma^2$, the covariance matrix shares the same eigenvectors
\begin{equation}
    \Sigma_{denoised}=Udiag(\gamma_k)U^T
\end{equation}
for
\begin{equation}
    \gamma_k=\frac{\lambda_k+2\alpha}{(\lambda_x+\alpha)^2}\Leftrightarrow\lambda_k=\frac{(1-2\alpha\gamma_k)+\sqrt{1+4\alpha\gamma_k}}{2\gamma_k}
\end{equation}

\hfill \break
Under this setting, we can come up with the following algorithm
\begin{enumerate}
    \item Fix $\alpha=\sigma^{-2}$
    \item Sample $\{x_n\}_{n=1,...,N}\sim q_\sigma$
    \item Construct sample covariance matrix $\hat{\Sigma}^2_{denoised}$ and sample mean $\hat{\mu}_{denoised}$
    \item Perform eigen-decomposition to acquire $\hat{U}$ and $\hat{\gamma}_k$, and eventually $\hat{\lambda}_k$
    \item Use them to compute $\hat{\mu}$ and $\hat{\Sigma}$
    \item Compute $m_i^{(p)}$
\end{enumerate}
